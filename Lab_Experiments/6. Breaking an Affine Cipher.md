## 6. Breaking an Affine Cipher

## CODE

#include <stdio.h>
#include <string.h>

int gcd(int a, int b)
{

    while (b != 0)
    {
    
        int t = b;
        b = a % b;
        a = t;
    }
    return a;
}

int inverse(int a)
{

    for (int i = 1; i < 26; i++)
        if ((a * i) % 26 == 1)
            return i;

    return -1;
}

int main()
{

    char cipher[500];
    int a, b, inv, i;

    printf("Enter ciphertext: ");
    fgets(cipher, sizeof(cipher), stdin);

    printf("\nPossible plaintexts:\n");

    for (a = 1; a < 26; a++)
    {
        if (gcd(a, 26) != 1)
            continue;

        inv = inverse(a);

        for (b = 0; b < 26; b++)
        {
            printf("\nKey a=%d, b=%d: ", a, b);

            for (i = 0; cipher[i] != '\0'; i++)
            {
                if (cipher[i] >= 'A' && cipher[i] <= 'Z')
                    printf("%c",
                    (inv * ((cipher[i] - 'A') - b + 26)) % 26 + 'A');

                else if (cipher[i] >= 'a' && cipher[i] <= 'z')
                    printf("%c",
                    (inv * ((cipher[i] - 'a') - b + 26)) % 26 + 'a');

                else
                    printf("%c", cipher[i]);
            }
        }
    }

    return 0;
}
