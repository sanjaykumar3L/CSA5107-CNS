# 24. RSA Private Key

## Program

#include <stdio.h>
#include <math.h>

// Extended Euclidean Algorithm to find modular inverse
int extended_gcd(int a, int b, int *x, int *y) {
    if (b == 0) {
        *x = 1;
        *y = 0;
        return a;
    }
    int x1, y1;
    int gcd = extended_gcd(b, a % b, &x1, &y1);
    *x = y1;
    *y = x1 - (a / b) * y1;
    return gcd;
}

int mod_inverse(int e, int phi) {
    int x, y;
    int gcd = extended_gcd(e, phi, &x, &y);
    if (gcd != 1) return -1; // Inverse does not exist
    return (x % phi + phi) % phi;
}

long long mod_exp(long long base, long long exp, long long mod) {
    long long res = 1;
    base %= mod;
    while (exp > 0) {
        if (exp % 2 == 1) res = (res * base) % mod;
        base = (base * base) % mod;
        exp /= 2;
    }
    return res;
}

int main() {
    int e = 31;
    int n = 3599;

    printf("=== RSA Private Key Derivation ===\n");
    printf("Public Key: e = %d, n = %d\n\n", e, n);

    // Step 1: Factorize n into p and q using trial division
    int p = 0, q = 0;
    int limit = (int)sqrt(n);
    for (int i = 2; i <= limit; i++) {
        if (n % i == 0) {
            p = i;
            q = n / i;
            break;
        }
    }

    if (p == 0) {
        printf("Failed to factorize n.\n");
        return 1;
    }

    printf("Step 1: Factorization of n = %d\n", n);
    printf("  Found prime factors: p = %d, q = %d\n", p, q);
    printf("  Check: %d * %d = %d\n\n", p, q, p * q);

    // Step 2: Calculate Euler's totient function phi(n)
    int phi = (p - 1) * (q - 1);
    printf("Step 2: Calculate phi(n)\n");
    printf("  phi(n) = (%d - 1) * (%d - 1) = %d * %d = %d\n\n", p, q, p - 1, q - 1, phi);

    // Step 3: Compute private exponent d using Extended Euclidean Algorithm
    int d = mod_inverse(e, phi);
    printf("Step 3: Multiplicative inverse of e (%d) mod phi (%d) using Extended Euclidean Algorithm\n", e, phi);
    printf("  Private key d = %d\n", d);
    printf("  Verification: (%d * %d) mod %d = %lld\n\n", e, d, phi, ((long long)e * d) % phi);

    // Step 4: Verification with encryption and decryption
    long long m = 1234; // Plaintext block
    long long c = mod_exp(m, e, n);
    long long decrypted = mod_exp(c, d, n);

    printf("=== Verification Test ===\n");
    printf("Original Plaintext (m) : %lld\n", m);
    printf("Encrypted Ciphertext (c): %lld\n", c);
    printf("Decrypted Plaintext     : %lld (%s)\n", decrypted, decrypted == m ? "SUCCESS" : "FAILURE");

    return 0;
}
