# 30. CBC MAC Forgery Attack

## Program

#include <stdio.h>
#include <stdint.h>

// Simple 64-bit reversible block cipher mock
uint64_t encrypt_block(uint64_t m, uint64_t k) {
    return ((m ^ k) * 0x5851F42D4C957F2DULL) ^ 0x14057B7EF767814FULL;
}

uint64_t cbc_mac_1block(uint64_t x, uint64_t k) {
    return encrypt_block(x, k);
}

uint64_t cbc_mac_2block(uint64_t b1, uint64_t b2, uint64_t k) {
    uint64_t c1 = encrypt_block(b1, k);
    uint64_t c2 = encrypt_block(b2 ^ c1, k);
    return c2;
}

int main() {
    uint64_t key = 0x2B7E151628AED2A6ULL;
    uint64_t X = 0x48656C6C6F313233ULL; // "Hello123"

    printf("=== CBC-MAC Extension Forgery Attack Demonstration ===\n");
    printf("Secret Key: 0x%016llX\n\n", key);

    // Step 1: Legitimate MAC for 1-block message X
    uint64_t T = cbc_mac_1block(X, key);
    printf("Step 1: Obtain MAC for one-block message X\n");
    printf("  Message X = 0x%016llX (\"Hello123\")\n", X);
    printf("  T = MAC(K, X) = 0x%016llX\n\n", T);

    // Step 2: Adversary computes forged block: X ^ T
    uint64_t forged_block2 = X ^ T;
    printf("Step 2: Adversary constructs two-block message M' = X || (X ^ T) without knowing K:\n");
    printf("  Block 1: X = 0x%016llX\n", X);
    printf("  Block 2: (X ^ T) = 0x%016llX\n\n", forged_block2);

    // Step 3: Compute MAC of M'
    uint64_t forged_MAC = cbc_mac_2block(X, forged_block2, key);
    printf("Step 3: Verification of CBC-MAC on forged message M':\n");
    printf("  Round 1: C1 = E_K(X) = T = 0x%016llX\n", T);
    printf("  Round 2: In = Block 2 ^ C1 = (X ^ T) ^ T = X\n");
    printf("  Final MAC = E_K(X) = T = 0x%016llX\n\n", forged_MAC);

    printf("Result: MAC(K, X || (X ^ T)) == MAC(K, X) (FORGERY %s!)\n", 
           forged_MAC == T ? "SUCCESSFUL" : "FAILED");
    printf("Vulnerability: Basic CBC-MAC is insecure for variable-length messages.\n");

    return 0;
}
