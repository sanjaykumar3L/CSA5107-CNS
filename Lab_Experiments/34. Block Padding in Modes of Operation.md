# 34. Block Padding in Modes of Operation

## Program

#include <stdio.h>
#include <string.h>
#include <stdint.h>

#define BLOCK_SIZE 8

int pad_data(const uint8_t *in, int len, uint8_t *out) {
    memcpy(out, in, len);
    int pad_len = BLOCK_SIZE - (len % BLOCK_SIZE);
    if (pad_len == 0) pad_len = BLOCK_SIZE;

    out[len] = 0x80;
    for (int i = 1; i < pad_len; i++) {
        out[len + i] = 0x00;
    }
    return len + pad_len;
}

int unpad_data(const uint8_t *in, int len, uint8_t *out) {
    int i = len - 1;
    while (i >= 0 && in[i] == 0x00) i--;
    if (i < 0 || in[i] != 0x80) return -1;
    memcpy(out, in, i);
    return i;
}

int main() {
    printf("=== Block Padding in ECB, CBC, and CFB Modes ===\n");
    printf("Block Size: 8 Bytes (64 bits). Padding: 1-bit followed by zeros (0x80 ... 0x00)\n\n");

    // Test 1: Incomplete block
    uint8_t t1[] = "Confidential!";
    int len1 = strlen((char*)t1);
    uint8_t p1[64], u1[64];
    int pad_len1 = pad_data(t1, len1, p1);
    int u_len1 = unpad_data(p1, pad_len1, u1);

    printf("Test 1: Plaintext length = %d bytes (Incomplete final block)\n", len1);
    printf("  Original Length : %d bytes\n", len1);
    printf("  Padded Length   : %d bytes (3 bytes padding: 80 00 00)\n", pad_len1);
    printf("  Unpadded Length : %d bytes (Integrity: %s)\n\n", u_len1, (len1 == u_len1) ? "OK" : "ERR");

    // Test 2: Full block
    uint8_t t2[] = "1234567812345678";
    int len2 = 16;
    uint8_t p2[64], u2[64];
    int pad_len2 = pad_data(t2, len2, p2);
    int u_len2 = unpad_data(p2, pad_len2, u2);

    printf("Test 2: Plaintext length = %d bytes (Already exact multiple of block size)\n", len2);
    printf("  Original Length : %d bytes\n", len2);
    printf("  Padded Length   : %d bytes (Appended full extra 8-byte padding block: 80 00 00 00 00 00 00 00)\n", pad_len2);
    printf("  Unpadded Length : %d bytes (Integrity: %s)\n\n", u_len2, (len2 == u_len2) ? "OK" : "ERR");

    printf("=== Motivation for Unconditional Extra Padding Block ===\n");
    printf("If messages ending exactly on block boundaries were not padded, the receiver cannot\n");
    printf("distinguish between real plaintext ending in 0x80 0x00 and added padding.\n");
    printf("By unconditionally padding every message, ambiguity is 100%% eliminated.\n");

    return 0;
}
