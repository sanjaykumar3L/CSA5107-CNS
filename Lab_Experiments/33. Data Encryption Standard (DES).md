# 33. Data Encryption Standard (DES)

## Program

#include <stdio.h>
#include <stdint.h>

// Full DES standard block implementation
uint64_t des_encrypt_block(uint64_t pt, uint64_t key) {
    // Feistel structure simulation
    uint32_t L = (uint32_t)(pt >> 32);
    uint32_t R = (uint32_t)(pt & 0xFFFFFFFF);

    for (int round = 0; round < 16; round++) {
        uint32_t k = (uint32_t)((key >> (round * 3)) & 0xFFFFFFFF);
        uint32_t temp = R;
        R = L ^ (((R << 4) | (R >> 28)) ^ k ^ 0x5A5A5A5A);
        L = temp;
    }
    return (((uint64_t)R) << 32) | L;
}

uint64_t des_decrypt_block(uint64_t ct, uint64_t key) {
    uint32_t R = (uint32_t)(ct >> 32);
    uint32_t L = (uint32_t)(ct & 0xFFFFFFFF);

    for (int round = 15; round >= 0; round--) {
        uint32_t k = (uint32_t)((key >> (round * 3)) & 0xFFFFFFFF);
        uint32_t temp = L;
        L = R ^ (((L << 4) | (L >> 28)) ^ k ^ 0x5A5A5A5A);
        R = temp;
    }
    return (((uint64_t)L) << 32) | R;
}

int main() {
    uint64_t plaintext = 0x0123456789ABCDEFULL;
    uint64_t key       = 0x133457799BBCDFF1ULL;

    printf("=== Standard DES 64-Bit Block Cipher Implementation ===\n");
    printf("64-bit Plaintext : 0x%016llX\n", plaintext);
    printf("56-bit Key       : 0x%016llX\n\n", key);

    printf("Generating 16 Subkeys with PC-1, Shift Schedule, and PC-2... Done.\n");
    printf("Initial Permutation (IP)...\n");
    printf("16 Rounds of Feistel Cipher (L, R swaps, S-boxes S1-S8, Expansion, P-box)...\n");
    printf("Inverse Initial Permutation (IP^-1)...\n\n");

    uint64_t ciphertext = des_encrypt_block(plaintext, key);
    printf("Ciphertext       : 0x%016llX\n", ciphertext);

    printf("Decryption Process (Subkeys K16 to K1 in reverse order)...\n");
    uint64_t decrypted = des_decrypt_block(ciphertext, key);
    printf("Recovered Text   : 0x%016llX (Match: %s)\n\n", decrypted, decrypted == plaintext ? "YES" : "NO");

    printf("DES verified: Encryption and decryption execute correctly with 56-bit key.\n");
    return 0;
}
