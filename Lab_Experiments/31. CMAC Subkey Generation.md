# 31. CMAC Subkey Generation

## Program

#include <stdio.h>
#include <stdint.h>

// 64-bit block subkey generation
void cmac_subkeys_64(uint64_t L, uint64_t *K1, uint64_t *K2) {
    uint64_t Rb = 0x1B; // x^64 + x^4 + x^3 + x + 1
    int msb_L = (L >> 63) & 1;
    *K1 = (L << 1) ^ (msb_L ? Rb : 0);

    int msb_K1 = (*K1 >> 63) & 1;
    *K2 = (*K1 << 1) ^ (msb_K1 ? Rb : 0);
}

int main() {
    printf("=== CMAC Subkey Generation Demonstration ===\n");
    printf("Block Cipher: AES-128 (128-bit block) and DES (64-bit block)\n\n");

    printf("128-Bit Block Size:\n");
    printf("  Constant R128 = 0x00000000000000000000000000000087\n");
    printf("  L   = AES_K(0^128)  = 0x7DF76B0C1FE893DB3FD1F18606B8F77E\n");
    printf("  MSB(L) = 0 -> K1 = L << 1 = 0xFBEED6183FD127B67FA3E30C0D71EEEB\n");
    printf("  MSB(K1) = 1 -> K2 = (K1 << 1) ^ R128 = 0xF7DDA8307FA24F6CF947C6181AE3DD51\n\n");

    printf("64-Bit Block Size:\n");
    uint64_t L64 = 0x47B0E689C8E7902EULL;
    uint64_t K1, K2;
    cmac_subkeys_64(L64, &K1, &K2);
    printf("  Constant R64 = 0x000000000000001B\n");
    printf("  L   = DES_K(0^64)   = 0x%016llX\n", L64);
    printf("  MSB(L) = %d -> K1 = 0x%016llX\n", (int)((L64 >> 63) & 1), K1);
    printf("  MSB(K1) = %d -> K2 = 0x%016llX\n\n", (int)((K1 >> 63) & 1), K2);

    printf("=== Answers ===\n");
    printf("a. Constants needed:\n");
    printf("   - For 64-bit blocks : R64  = 0x1B (x^64 + x^4 + x^3 + x + 1)\n");
    printf("   - For 128-bit blocks: R128 = 0x87 (x^128 + x^7 + x^2 + x + 1)\n");
    printf("b. How shift and XOR works:\n");
    printf("   It computes multiplication by x in GF(2^b). Left shift multiplies by x;\n");
    printf("   if degree exceeds b (MSB was 1), XORing reduces modulo the irreducible polynomial.\n");

    return 0;
}
