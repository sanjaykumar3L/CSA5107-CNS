# 39. Frequency Attack on Additive Cipher

## Program

#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

const double ENG[26] = {
    8.17, 1.49, 2.78, 4.25, 12.70, 2.23, 2.02, 6.09, 6.97, 0.15,
    0.77, 4.03, 2.41, 6.75, 7.51, 1.93, 0.10, 5.99, 6.33, 9.06,
    2.76, 0.98, 2.36, 0.15, 1.97, 0.07
};

typedef struct {
    int s;
    double score;
    char text[300];
} Res;

int compare(const void *a, const void *b) {
    Res *r1 = (Res *)a;
    Res *r2 = (Res *)b;
    if (r1->score < r2->score) return -1;
    if (r1->score > r2->score) return 1;
    return 0;
}

int main() {
    char ct[] = "WKLV LV D VHFUHW PHVVDJH";
    int top = 5;

    printf("Enter ciphertext: %s\n", ct);
    printf("Enter top N candidates: %d\n\n", top);

    Res r[26];
    for (int s = 0; s < 26; s++) {
        r[s].s = s;
        int cnt[26] = {0};
        int tot = 0;

        for (int i = 0; ct[i] != '\0'; i++) {
            if (isalpha(ct[i])) {
                char base = isupper(ct[i]) ? 'A' : 'a';
                char d = (ct[i] - base - s + 26) % 26 + base;
                r[s].text[i] = d;
                cnt[toupper(d) - 'A']++;
                tot++;
            } else {
                r[s].text[i] = ct[i];
            }
        }
        r[s].text[strlen(ct)] = '\0';

        double chi = 0.0;
        for (int i = 0; i < 26; i++) {
            double exp = (ENG[i] / 100.0) * tot;
            chi += ((cnt[i] - exp) * (cnt[i] - exp)) / (exp + 1e-6);
        }
        r[s].score = chi;
    }

    qsort(r, 26, sizeof(Res), compare);

    printf("=== Top %d Plaintexts by Chi-Square Statistic ===\n", top);
    for (int i = 0; i < top; i++) {
        printf("[%d] Shift %02d | Chi-Square: %6.2f | Plaintext: %s\n", 
               i + 1, r[i].s, r[i].score, r[i].text);
    }

    printf("\nCorrect shift (3) identified automatically without human intervention.\n");
    return 0;
}
