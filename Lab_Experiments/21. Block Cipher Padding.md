# 21. Block Cipher Padding

## Program

#include <stdio.h>
#include <string.h>
#include <stdint.h>

#define BLOCK_SIZE 8 // 64 bits = 8 bytes

// Pad a message using 1-bit followed by 0-bits (0x80 followed by 0x00 bytes)
// Always appends padding even if length is already a multiple of block size.
int pad_message(const uint8_t *in, int in_len, uint8_t *out) {

    memcpy(out, in, in_len);
    int pad_len = BLOCK_SIZE - (in_len % BLOCK_SIZE);
    if (pad_len == 0) {
        pad_len = BLOCK_SIZE; // Append full padding block
    }
    
    // First padding byte has 1 bit in MSB (0x80), rest 0s
    out[in_len] = 0x80;
    for (int i = 1; i < pad_len; i++) {
        out[in_len + i] = 0x00;
    }
    return in_len + pad_len;
}

// Unpad message by stripping trailing 0x00 bytes and the final 0x80 byte
int unpad_message(const uint8_t *in, int padded_len, uint8_t *out) {

    int i = padded_len - 1;
    while (i >= 0 && in[i] == 0x00) {
        i--;
    }
    if (i < 0 || in[i] != 0x80) {
        return -1; // Invalid padding
    }
    int orig_len = i;
    memcpy(out, in, orig_len);
    return orig_len;
}

void print_hex(const char *label, const uint8_t *data, int len) {

    printf("%-26s (%2d bytes): ", label, len);
    for (int i = 0; i < len; i++) {
        printf("%02X ", data[i]);
    }
    printf("\n");
}

int main() {

    printf("=== Demonstration of Unconditional Block Cipher Padding (1-bit followed by zeros) ===\n\n");

    // Case 1: Incomplete block (11 bytes: 1 full block + 3 bytes)
    uint8_t msg1[] = "Hello World"; // 11 bytes
    int len1 = strlen((char *)msg1);
    uint8_t padded1[64], unpadded1[64];

    int pad_len1 = pad_message(msg1, len1, padded1);
    int orig_len1 = unpad_message(padded1, pad_len1, unpadded1);

    printf("Case 1: Message length NOT a multiple of block size\n");
    print_hex("Original Message", msg1, len1);
    print_hex("Padded Message", padded1, pad_len1);
    print_hex("Recovered Message", unpadded1, orig_len1);
    printf("Integrity check: %s\n\n", (len1 == orig_len1 && memcmp(msg1, unpadded1, len1) == 0) ? "SUCCESS" : "FAILED");

    // Case 2: Exactly complete blocks (16 bytes: 2 full 8-byte blocks)
    uint8_t msg2[] = "1234567812345678"; // 16 bytes
    int len2 = 16;
    uint8_t padded2[64], unpadded2[64];

    int pad_len2 = pad_message(msg2, len2, padded2);
    int orig_len2 = unpad_message(padded2, pad_len2, unpadded2);

    printf("Case 2: Message length ALREADY an exact multiple of block size\n");
    print_hex("Original Message", msg2, len2);
    print_hex("Padded (extra block)", padded2, pad_len2);
    print_hex("Recovered Message", unpadded2, orig_len2);
    printf("Integrity check: %s\n\n", (len2 == orig_len2 && memcmp(msg2, unpadded2, len2) == 0) ? "SUCCESS" : "FAILED");

    printf("=== Motivation for Unconditional Padding ===\n");
    printf("Question: What is the motivation for including a padding block when padding is not needed?\n\n");
    printf("Answer:\n");
    printf("Without mandatory padding, if a message naturally ends with bits matching the padding pattern\n");
    printf("(e.g., bit 1 followed by zeros), the receiver has no way to distinguish whether those bits\n");
    printf("were part of the original plaintext or added as padding.\n\n");
    printf("By requiring EVERY message to be padded (even if already a full multiple, where an entire extra\n");
    printf("padding block is appended), the receiver knows unambiguously that padding is ALWAYS present,\n");
    printf("can safely scan backwards from the last bit to locate the delimiter '1' bit, strip the padding,\n");
    printf("and recover the exact original message without ambiguity or data corruption.\n");

    return 0;
}
