## 12. Hill Cipher — Encrypt and Decrypt

## CODE

#include <stdio.h>
#include <string.h>
#include <ctype.h>

int inverse(int a)
{

    for (int i = 1; i < 26; i++)
        if ((a * i) % 26 == 1)
            return i;

    return -1;
}

int main()
{

    char text[200];
    int key[2][2] = {{9, 4}, {5, 7}};
    int invKey[2][2];
    int det, invDet;

    printf("Enter plaintext: ");
    fgets(text, sizeof(text), stdin);

    det = key[0][0] * key[1][1] -
          key[0][1] * key[1][0];

    det = (det % 26 + 26) % 26;
    invDet = inverse(det);

    invKey[0][0] =  key[1][1] * invDet % 26;
    invKey[0][1] = -key[0][1] * invDet % 26;
    invKey[1][0] = -key[1][0] * invDet % 26;
    invKey[1][1] =  key[0][0] * invDet % 26;

    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            invKey[i][j] = (invKey[i][j] + 26) % 26;

    printf("Ciphertext: ");

    int nums[200], n = 0;

    for (int i = 0; text[i] != '\0'; i++)
        if (isalpha(text[i]))
            nums[n++] = toupper(text[i]) - 'A';

    if (n % 2 != 0)
        nums[n++] = 'X' - 'A';

    for (int i = 0; i < n; i += 2)
    {
        int x = (key[0][0] * nums[i] +
                 key[0][1] * nums[i + 1]) % 26;

        int y = (key[1][0] * nums[i] +
                 key[1][1] * nums[i + 1]) % 26;

        printf("%c%c", x + 'A', y + 'A');
    }

    printf("\n");

    return 0;
}
