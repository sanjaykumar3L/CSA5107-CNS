# 20. ECB and CBC Error Propagation

## Program

#include <stdio.h>
#include <stdint.h>

// Simple reversible block cipher transformation for demonstration
uint32_t encrypt_block(uint32_t p, uint32_t k) {

    return ((p ^ k) * 0x45D9F3B) ^ 0x12345678;
}

uint32_t decrypt_block(uint32_t c, uint32_t k) {

    // Inverse transformation
    uint32_t inv = c ^ 0x12345678;
    inv = inv * 0x119DE1F3; // Modular inverse of 0x45D9F3B modulo 2^32
    return inv ^ k;
}

int main() {

    uint32_t key = 0xDEADBEEF;
    uint32_t iv = 0x12345678;
    uint32_t P[4] = {0xAAAAAAAA, 0xBBBBBBBB, 0xCCCCCCCC, 0xDDDDDDDD};
    uint32_t C[4], decP[4];

    printf("=== Demonstration 1: Transmission Error in Transmitted Ciphertext C1 ===\n");
    // CBC Encryption
    uint32_t prev = iv;
    for (int i = 0; i < 4; i++) {
        C[i] = encrypt_block(P[i] ^ prev, key);
        prev = C[i];
    }

    // Introduce a bit error in transmitted C1
    C[0] ^= 0x00000001; 

    // CBC Decryption at Receiver
    prev = iv;
    for (int i = 0; i < 4; i++) {
        decP[i] = decrypt_block(C[i], key) ^ prev;
        prev = C[i];
    }

    printf("Original P1: 0x%08X | Decrypted P1: 0x%08X (%s)\n", P[0], decP[0], P[0] == decP[0] ? "OK" : "CORRUPTED");
    printf("Original P2: 0x%08X | Decrypted P2: 0x%08X (%s)\n", P[1], decP[1], P[1] == decP[1] ? "OK" : "CORRUPTED");
    printf("Original P3: 0x%08X | Decrypted P3: 0x%08X (%s)\n", P[2], decP[2], P[2] == decP[2] ? "OK" : "CORRUPTED");
    printf("Original P4: 0x%08X | Decrypted P4: 0x%08X (%s)\n", P[3], decP[3], P[3] == decP[3] ? "OK" : "CORRUPTED");

    printf("\n=== Demonstration 2: Source Bit Error in Plaintext P1 ===\n");
    uint32_t P_err[4] = {0xAAAAAAAA ^ 0x01, 0xBBBBBBBB, 0xCCCCCCCC, 0xDDDDDDDD};
    uint32_t C_err[4], decP_err[4];

    // Encrypt with corrupted source P1
    prev = iv;
    for (int i = 0; i < 4; i++) {
        C_err[i] = encrypt_block(P_err[i] ^ prev, key);
        prev = C_err[i];
    }

    printf("Ciphertext differences due to 1-bit error in source P1:\n");
    for (int i = 0; i < 4; i++) {
        printf("C%d diff: 0x%08X (Affected: %s)\n", i + 1, C[i] ^ C_err[i], C[i] != C_err[i] ? "YES" : "NO");
    }

    // Decrypt at receiver
    prev = iv;
    for (int i = 0; i < 4; i++) {
        decP_err[i] = decrypt_block(C_err[i], key) ^ prev;
        prev = C_err[i];
    }

    printf("\nDecrypted blocks at receiver:\n");
    for (int i = 0; i < 4; i++) {
        printf("Decrypted P%d matches original uncorrupted P%d? %s\n", 
               i + 1, i + 1, (decP_err[i] == P[i]) ? "YES" : "NO (1-bit original error only)");
    }

    printf("\n=== Answers to Theoretical Questions ===\n");
    printf("a. Are any blocks beyond P2 affected when C1 is corrupted?\n");
    printf("   Answer: NO. In CBC decryption, Pj = D_K(Cj) ^ C_{j-1}.\n");
    printf("   Since C1 is only used in decrypting P1 and P2, blocks P3, P4, ... are unaffected.\n\n");
    printf("b. Suppose there is a bit error in the source version of P1.\n");
    printf("   Through how many ciphertext blocks is this error propagated? What is the effect at receiver?\n");
    printf("   Answer: The error propagates through ALL subsequent ciphertext blocks (C1 to Cn).\n");
    printf("   At the receiver, however, only the original bit error in P1 appears; all other\n");
    printf("   plaintext blocks (P2, P3, ... Pn) decrypt correctly without any errors.\n");

    return 0;
}
