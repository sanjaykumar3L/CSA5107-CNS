# 18. DES Subkey Generation

## Program

#include <stdio.h>
#include <stdint.h>

// Standard DES Permuted Choice 2 (PC-2) table (48 bits from 56 bits)
const int PC2[48] = {
    14, 17, 11, 24,  1,  5,
     3, 28, 15,  6, 21, 10,
    23, 19, 12,  4, 26,  8,
    16,  7, 27, 20, 13,  2,
    41, 52, 31, 37, 47, 55,
    30, 40, 51, 45, 33, 48,
    44, 49, 39, 56, 34, 53,
    46, 42, 50, 36, 29, 32
};

int main() {
    printf("=== Analysis of DES Permuted Choice 2 (PC-2) Subkey Generation ===\n\n");
    printf("Verifying bit selection from two disjoint 28-bit halves (C and D):\n");
    printf(" - Half C covers bit indices 1 to 28\n");
    printf(" - Half D covers bit indices 29 to 56\n\n");

    int c_count = 0, d_count = 0;
    int first_half_valid = 1;
    int second_half_valid = 1;

    printf("First 24 subkey bit sources (from PC-2):\n");
    for (int i = 0; i < 24; i++) {
        int bit_src = PC2[i];
        printf("%2d ", bit_src);
        if ((i + 1) % 6 == 0) printf("\n");
        if (bit_src >= 1 && bit_src <= 28) c_count++;
        else first_half_valid = 0;
    }

    printf("\nSecond 24 subkey bit sources (from PC-2):\n");
    for (int i = 24; i < 48; i++) {
        int bit_src = PC2[i];
        printf("%2d ", bit_src);
        if ((i + 1) % 6 == 0) printf("\n");
        if (bit_src >= 29 && bit_src <= 56) d_count++;
        else second_half_valid = 0;
    }

    printf("\n=== Verification Results ===\n");
    printf("Total bits chosen for 1st half of subkey from C (1-28): %d / 24\n", c_count);
    printf("Total bits chosen for 2nd half of subkey from D (29-56): %d / 24\n", d_count);

    if (first_half_valid && second_half_valid) {
        printf("\nCONCLUSION:\n");
        printf("PROVEN: The first 24 bits of each subkey come strictly from the 28-bit subset C,\n");
        printf("and the second 24 bits come strictly from the disjoint 28-bit subset D.\n");
        printf("No bits from C cross over to the second 24 bits, and no bits from D cross into the first 24 bits.\n");
    } else {
        printf("Property verification failed.\n");
    }

    return 0;
}
