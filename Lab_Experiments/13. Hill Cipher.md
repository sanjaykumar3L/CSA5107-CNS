## 13. Hill Cipher — Known Plaintext Attack

## CODE

#include <stdio.h>

int inverse(int a)
{

    for (int i = 1; i < 26; i++)
        if ((a * i) % 26 == 1)
            return i;

    return -1;
}

int main()
{

    int P[2][2], C[2][2];
    int inv[2][2], K[2][2];
    int det, invDet;

    printf("Enter plaintext matrix:\n");

    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            scanf("%d", &P[i][j]);

    printf("Enter ciphertext matrix:\n");

    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            scanf("%d", &C[i][j]);

    det = P[0][0] * P[1][1] -
          P[0][1] * P[1][0];

    det = (det % 26 + 26) % 26;
    invDet = inverse(det);

    if (invDet == -1)
    {
        printf("Matrix cannot be inverted.\n");
        return 0;
    }

    inv[0][0] = P[1][1] * invDet;
    inv[0][1] = -P[0][1] * invDet;
    inv[1][0] = -P[1][0] * invDet;
    inv[1][1] = P[0][0] * invDet;

    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            inv[i][j] = (inv[i][j] % 26 + 26) % 26;

    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
        {
            K[i][j] = 0;

            for (int k = 0; k < 2; k++)
                K[i][j] += C[i][k] * inv[k][j];

            K[i][j] %= 26;
        }

    printf("\nRecovered Key Matrix:\n");

    for (int i = 0; i < 2; i++)
    {
        for (int j = 0; j < 2; j++)
            printf("%d ", K[i][j]);

        printf("\n");
    }

    return 0;
}
