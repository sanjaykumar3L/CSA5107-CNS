# 40. Automated Frequency Attack on Monoalphabetic Cipher

## Program

#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

const double ENG_FREQ[26] = {
    8.17, 1.49, 2.78, 4.25, 12.70, 2.23, 2.02, 6.09, 6.97, 0.15,
    0.77, 4.03, 2.41, 6.75, 7.51, 1.93, 0.10, 5.99, 6.33, 9.06,
    2.76, 0.98, 2.36, 0.15, 1.97, 0.07
};

typedef struct {
    int shift;
    double score;
    char text[500];
} Candidate;

int compare_candidates(const void *a, const void *b) {
    Candidate *c1 = (Candidate *)a;
    Candidate *c2 = (Candidate *)b;
    if (c1->score < c2->score) return -1;
    if (c1->score > c2->score) return 1;
    return 0;
}

int main() {
    char ct[500] = "KHOOR ZRUOG WKLV LV D VHFUHW PHVVDJH";
    int top = 10;

    printf("Enter ciphertext: %s\n", ct);
    printf("Enter top candidates: %d\n\n", top);

    Candidate cands[26];
    for (int s = 0; s < 26; s++) {
        cands[s].shift = s;
        int counts[26] = {0};
        int tot = 0;

        for (int i = 0; ct[i] != '\0'; i++) {
            if (isalpha(ct[i])) {
                char base = isupper(ct[i]) ? 'A' : 'a';
                char d = (ct[i] - base - s + 26) % 26 + base;
                cands[s].text[i] = d;
                counts[toupper(d) - 'A']++;
                tot++;
            } else {
                cands[s].text[i] = ct[i];
            }
        }
        cands[s].text[strlen(ct)] = '\0';

        double chi = 0.0;
        for (int i = 0; i < 26; i++) {
            double exp = (ENG_FREQ[i] / 100.0) * tot;
            chi += ((counts[i] - exp) * (counts[i] - exp)) / (exp + 1e-6);
        }
        cands[s].score = chi;
    }

    qsort(cands, 26, sizeof(Candidate), compare_candidates);

    printf("=== Top %d Most Likely Plaintexts ===\n", top);
    for (int i = 0; i < top; i++) {
        printf("[%2d] (Shift: %2d, Score: %7.2f): %s\n", 
               i + 1, cands[i].shift, cands[i].score, cands[i].text);
    }

    return 0;
}
