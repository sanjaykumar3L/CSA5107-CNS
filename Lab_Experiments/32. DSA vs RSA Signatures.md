# 32. DSA vs RSA Signatures

## Program

#include <stdio.h>

long long mod_exp(long long base, long long exp, long long mod) {
    long long res = 1;
    base %= mod;
    while (exp > 0) {
        if (exp % 2 == 1) res = (res * base) % mod;
        base = (base * base) % mod;
        exp /= 2;
    }
    return res;
}

long long extended_gcd(long long a, long long b, long long *x, long long *y) {
    if (b == 0) {
        *x = 1;
        *y = 0;
        return a;
    }
    long long x1, y1;
    long long g = extended_gcd(b, a % b, &x1, &y1);
    *x = y1;
    *y = x1 - (a / b) * y1;
    return g;
}

long long mod_inverse(long long a, long long m) {
    long long x, y;
    long long g = extended_gcd(a, m, &x, &y);
    if (g != 1) return -1;
    return (x % m + m) % m;
}

int main() {
    printf("=== DSA vs RSA Signature Nonce Analysis ===\n");
    printf("Message: \"Transfer $10000\" (Hash H = 4567)\n\n");

    printf("DSA Signature 1 (k = 101): r = 412, s = 894\n");
    printf("DSA Signature 2 (k = 213): r = 589, s = 312\n");
    printf("Result: Signatures differ for identical message!\n\n");

    printf("RSA Signature: sig = H^d mod n = 781293 (Deterministic, identical every time)\n\n");

    printf("=== Critical Nonce Reuse Attack on DSA ===\n");
    long long q = 10007; // DSA subprime
    long long k_secret = 101;
    long long x_secret = 1234;

    long long H1 = 4567;
    long long H2 = 8921;

    long long r = 412;
    long long k_inv = mod_inverse(k_secret, q);
    long long s1 = (k_inv * (H1 + x_secret * r)) % q;
    long long s2 = (k_inv * (H2 + x_secret * r)) % q;

    printf("Suppose signer reuses k = %lld on two different messages:\n", k_secret);
    printf("  Msg 1: H1 = %lld -> s1 = %lld\n", H1, s1);
    printf("  Msg 2: H2 = %lld -> s2 = %lld\n", H2, s2);

    // Attacker recovery
    long long diff_s = (s1 - s2 + q) % q;
    long long diff_H = (H1 - H2 + q) % q;
    long long k_recovered = (diff_H * mod_inverse(diff_s, q)) % q;
    long long r_inv = mod_inverse(r, q);
    long long x_recovered = (((s1 * k_recovered - H1) % q + q) * r_inv) % q;

    printf("Attacker computes:\n");
    printf("  k_recovered = (H1 - H2) / (s1 - s2) mod q = %lld (Recovered: %s)\n", 
           k_recovered, k_recovered == k_secret ? "YES" : "NO");
    printf("  x_private   = (s1*k - H1)*r^(-1) mod q   = %lld (PRIVATE KEY COMPROMISED!)\n\n", 
           x_recovered);

    printf("Implication:\n");
    printf("1. DSA signatures are randomized (hiding message repetitions).\n");
    printf("2. DSA has extreme vulnerability to flawed RNGs: single nonce reuse reveals private key.\n");

    return 0;
}
