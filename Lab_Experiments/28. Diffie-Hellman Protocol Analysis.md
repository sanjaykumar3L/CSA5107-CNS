# 28. Diffie-Hellman Protocol Analysis

## Program

#include <stdio.h>

long long mod_exp(long long base, long long exp, long long mod) {
    long long res = 1;
    base %= mod;
    while (exp > 0) {
        if (exp % 2 == 1) res = (res * base) % mod;
        base = (base * base) % mod;
        exp /= 2;
    }
    return res;
}

long long extended_gcd(long long a, long long b, long long *x, long long *y) {
    if (b == 0) {
        *x = 1;
        *y = 0;
        return a;
    }
    long long x1, y1;
    long long g = extended_gcd(b, a % b, &x1, &y1);
    *x = y1;
    *y = x1 - (a / b) * y1;
    return g;
}

long long mod_inverse(long long a, long long m) {
    long long x, y;
    long long g = extended_gcd(a, m, &x, &y);
    if (g != 1) return -1;
    return (x % m + m) % m;
}

int main() {
    long long q = 353; // Public prime
    long long a = 11;  // Public exponent

    printf("=== Diffie-Hellman Variant: Sending x^a mod q instead of a^x mod q ===\n");
    printf("Public Parameters: prime q = %lld, public exponent a = %lld\n", q, a);

    long long x1 = 97;  // Alice's secret
    long long x2 = 125; // Bob's secret

    // Alice and Bob transmit x^a mod q
    long long Y1 = mod_exp(x1, a, q);
    long long Y2 = mod_exp(x2, a, q);

    printf("Alice secret: x1 = %lld, sends Y1 = (x1^a) mod q = %lld\n", x1, Y1);
    printf("Bob secret  : x2 = %lld, sends Y2 = (x2^a) mod q = %lld\n\n", x2, Y2);

    // Eve intercepts Y1 and Y2
    printf("Step 1: Attacker Eve intercepts Y1 = %lld, Y2 = %lld.\n", Y1, Y2);
    printf("Since q is prime, phi(q) = q - 1 = %lld is publicly known!\n", q - 1);

    long long d = mod_inverse(a, q - 1);
    printf("Eve computes inverse d = a^(-1) mod (q - 1):\n");
    printf("  d = %lld^(-1) mod %lld = %lld\n\n", a, q - 1, d);

    // Eve recovers x1 and x2
    long long recovered_x1 = mod_exp(Y1, d, q);
    long long recovered_x2 = mod_exp(Y2, d, q);

    printf("Step 2: Eve recovers the secrets directly:\n");
    printf("  Eve recovered x1 = %lld (Match: %s)\n", recovered_x1, recovered_x1 == x1 ? "YES" : "NO");
    printf("  Eve recovered x2 = %lld (Match: %s)\n", recovered_x2, recovered_x2 == x2 ? "YES" : "NO");

    printf("\n=== Answers to Protocol Questions ===\n");
    printf("1. Can Eve break the system without finding secrets? Yes, easily.\n");
    printf("2. Can Eve find the secret numbers? Yes! Because q - 1 is public, solving x^a mod q\n");
    printf("   is trivial using modular inverse of a modulo (q - 1), unlike discrete log a^x mod q.\n");

    return 0;
}
