# 36. Affine Caesar Cipher Validity

## Program

#include <stdio.h>

int gcd(int a, int b) {
    while (b != 0) {
        int t = b;
        b = a % b;
        a = t;
    }
    return a;
}

int main() {
    printf("=== Affine Caesar Cipher Validity Analysis ===\n");
    printf("Formula: C = (a * p + b) mod 26\n\n");

    printf("Testing Values of 'a' for Bijective (One-to-One) Mapping:\n");
    printf("Allowed values (gcd(a, 26) == 1):\n  [");
    int allowed = 0, disallowed = 0;
    for (int a = 0; a < 26; a++) {
        if (gcd(a, 26) == 1) {
            printf("%2d ", a);
            allowed++;
        }
    }
    printf("] (Total: %d values)\n\n", allowed);

    printf("Disallowed values (gcd(a, 26) != 1):\n  [");
    for (int a = 0; a < 26; a++) {
        if (gcd(a, 26) != 1) {
            printf("%2d ", a);
            disallowed++;
        }
    }
    printf("] (Total: %d values)\n\n", disallowed);

    printf("Demonstration of Failure for a = 2, b = 3:\n");
    int c1 = (2 * 0 + 3) % 26;
    int c2 = (2 * 13 + 3) % 26;
    printf("  p = 0  -> C = (2 * 0 + 3) mod 26 = %d\n", c1);
    printf("  p = 13 -> C = (2 * 13 + 3) mod 26 = %d (COLLISION: More than one p maps to C!)\n\n", c2);

    printf("=== Answers ===\n");
    printf("a. Limitations on b: None. Any b in [0, 25] produces a valid bijection.\n");
    printf("b. Values of 'a' not allowed: Any integer sharing a factor with 26 (multiples of 2 or 13).\n");

    return 0;
}
