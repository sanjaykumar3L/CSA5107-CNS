# 23. S-DES CTR Mode

## Program

#include <stdio.h>
#include <stdint.h>

// S-DES Tables
const int P10[] = {3, 5, 2, 7, 4, 10, 1, 9, 8, 6};
const int P8[]  = {6, 3, 7, 4, 8, 5, 10, 9};
const int IP[]  = {2, 6, 3, 1, 4, 8, 5, 7};
const int IP_INV[] = {4, 1, 3, 5, 7, 2, 8, 6};
const int EP[]  = {4, 1, 2, 3, 2, 3, 4, 1};
const int P4[]  = {2, 4, 3, 1};

const int S0[4][4] = {
    {1, 0, 3, 2},
    {3, 2, 1, 0},
    {0, 2, 1, 3},
    {3, 1, 3, 2}
};

const int S1[4][4] = {
    {0, 1, 2, 3},
    {2, 0, 1, 3},
    {3, 0, 1, 0},
    {2, 1, 0, 3}
};

uint16_t permute(uint16_t in, const int *table, int n_in, int n_out) {
    uint16_t out = 0;
    for (int i = 0; i < n_out; i++) {
        int bit = (in >> (n_in - table[i])) & 1;
        out = (out << 1) | bit;
    }
    return out;
}

uint8_t left_shift_5(uint8_t val, int shifts) {
    return ((val << shifts) | (val >> (5 - shifts))) & 0x1F;
}

void generate_subkeys(uint16_t key, uint8_t *k1, uint8_t *k2) {
    uint16_t p10 = permute(key, P10, 10, 10);
    uint8_t left = (p10 >> 5) & 0x1F;
    uint8_t right = p10 & 0x1F;

    left = left_shift_5(left, 1);
    right = left_shift_5(right, 1);
    *k1 = (uint8_t)permute((left << 5) | right, P8, 10, 8);

    left = left_shift_5(left, 2);
    right = left_shift_5(right, 2);
    *k2 = (uint8_t)permute((left << 5) | right, P8, 10, 8);
}

uint8_t f_k(uint8_t r, uint8_t sk) {
    uint8_t ep = (uint8_t)permute(r, EP, 4, 8);
    ep ^= sk;
    
    int row0 = (((ep >> 7) & 1) << 1) | ((ep >> 4) & 1);
    int col0 = (((ep >> 6) & 1) << 1) | ((ep >> 5) & 1);
    int s0_val = S0[row0][col0];

    int row1 = (((ep >> 3) & 1) << 1) | (ep & 1);
    int col1 = (((ep >> 2) & 1) << 1) | ((ep >> 1) & 1);
    int s1_val = S1[row1][col1];

    uint8_t combined = (s0_val << 2) | s1_val;
    return (uint8_t)permute(combined, P4, 4, 4);
}

uint8_t sdes_round(uint8_t block, uint8_t sk) {
    uint8_t left = (block >> 4) & 0x0F;
    uint8_t right = block & 0x0F;
    uint8_t f_res = f_k(right, sk);
    return ((left ^ f_res) << 4) | right;
}

uint8_t sdes_encrypt(uint8_t block, uint8_t k1, uint8_t k2) {
    block = (uint8_t)permute(block, IP, 8, 8);
    block = sdes_round(block, k1);
    block = ((block & 0x0F) << 4) | ((block >> 4) & 0x0F);
    block = sdes_round(block, k2);
    return (uint8_t)permute(block, IP_INV, 8, 8);
}

void print_bin8(uint8_t b) {
    for (int i = 7; i >= 0; i--) {
        printf("%d", (b >> i) & 1);
        if (i == 4) printf(" ");
    }
}

int main() {
    uint16_t key = 0b0111111101; // Key: 01111 11101
    uint8_t counter_start = 0b00000000; // Counter starting at 0000 0000
    
    // Plaintext: 0000 0001 0000 0010 0000 0100
    uint8_t p[3] = {0b00000001, 0b00000010, 0b00000100};
    uint8_t c[3];
    uint8_t dec[3];

    uint8_t k1, k2;
    generate_subkeys(key, &k1, &k2);

    printf("=== S-DES Counter (CTR) Mode Demonstration ===\n");
    printf("Binary Key    : 01111 11101\n");
    printf("Initial Counter: "); print_bin8(counter_start); printf("\n\n");

    // CTR Encryption: C_i = P_i ^ Encrypt(Counter_i)
    printf("CTR Encryption Process:\n");
    for (int i = 0; i < 3; i++) {
        uint8_t ctr_val = counter_start + i;
        uint8_t keystream = sdes_encrypt(ctr_val, k1, k2);
        c[i] = p[i] ^ keystream;
        printf("Block %d: CTR = ", i + 1); print_bin8(ctr_val);
        printf(" | Keystream = "); print_bin8(keystream);
        printf(" | Ciphertext = "); print_bin8(c[i]);
        printf("\n");
    }

    printf("\nTotal Plaintext : ");
    for (int i = 0; i < 3; i++) { print_bin8(p[i]); printf(" "); }
    printf("\nTotal Ciphertext: ");
    for (int i = 0; i < 3; i++) { print_bin8(c[i]); printf(" "); }
    printf("\n\n");

    // CTR Decryption: P_i = C_i ^ Encrypt(Counter_i)
    printf("CTR Decryption Process:\n");
    for (int i = 0; i < 3; i++) {
        uint8_t ctr_val = counter_start + i;
        uint8_t keystream = sdes_encrypt(ctr_val, k1, k2);
        dec[i] = c[i] ^ keystream;
        printf("Block %d: Decrypted = ", i + 1); print_bin8(dec[i]);
        printf("\n");
    }

    return 0;
}
