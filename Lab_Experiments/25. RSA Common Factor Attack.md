# 25. RSA Common Factor Attack

## Program

#include <stdio.h>
#include <stdint.h>

// Euclidean Algorithm for greatest common divisor
long long gcd(long long a, long long b) {
    while (b != 0) {
        long long t = b;
        b = a % b;
        a = t;
    }
    return a;
}

// Extended Euclidean Algorithm to find modular inverse
long long extended_gcd(long long a, long long b, long long *x, long long *y) {
    if (b == 0) {
        *x = 1;
        *y = 0;
        return a;
    }
    long long x1, y1;
    long long g = extended_gcd(b, a % b, &x1, &y1);
    *x = y1;
    *y = x1 - (a / b) * y1;
    return g;
}

long long mod_inverse(long long e, long long phi) {
    long long x, y;
    long long g = extended_gcd(e, phi, &x, &y);
    if (g != 1) return -1;
    return (x % phi + phi) % phi;
}

long long mod_exp(long long base, long long exp, long long mod) {
    long long res = 1;
    base %= mod;
    while (exp > 0) {
        if (exp % 2 == 1) res = (__int128)res * base % mod;
        base = (__int128)base * base % mod;
        exp /= 2;
    }
    return res;
}

int main() {
    // RSA Setup: Unknown private key to the attacker
    long long p_secret = 10007;
    long long q_secret = 10009;
    long long n = p_secret * q_secret; // 100160063
    long long e = 65537;

    // A set of message blocks
    // Block 1 is a multiple of p: gcd(M1, n) = p
    long long M[3] = {10007 * 3, 123456, 789012}; 
    long long C[3];

    // Encryption by sender
    for (int i = 0; i < 3; i++) {
        C[i] = mod_exp(M[i], e, n);
    }

    printf("=== RSA Common Factor Attack Demonstration ===\n");
    printf("Public Key: n = %lld, e = %lld\n", n, e);
    printf("Intercepted Ciphertext Blocks:\n");
    for (int i = 0; i < 3; i++) {
        printf("  C[%d] = %lld\n", i + 1, C[i]);
    }

    printf("\nAttacker Knowledge: One plaintext block shares a factor with n.\n");
    printf("Property: Since C = M^e mod n, if p divides M, then p also divides C.\n");
    printf("Therefore, gcd(C, n) will reveal the prime factor p!\n\n");

    // Attacker runs GCD on the ciphertext blocks against n
    long long p_found = 1, q_found = 1;
    for (int i = 0; i < 3; i++) {
        long long factor = gcd(C[i], n);
        if (factor > 1 && factor < n) {
            printf("[!] Success! Found non-trivial factor from C[%d]:\n", i + 1);
            p_found = factor;
            q_found = n / factor;
            printf("    p = %lld\n", p_found);
            printf("    q = %lld\n", q_found);
            break;
        }
    }

    if (p_found == 1) {
        printf("Attack failed: no common factor found.\n");
        return 1;
    }

    // Step 2: Compute phi(n) and recover private key d
    long long phi = (p_found - 1) * (q_found - 1);
    long long d = mod_inverse(e, phi);

    printf("\nPrivate Key Recovery:\n");
    printf("  phi(n) = (%lld - 1) * (%lld - 1) = %lld\n", p_found, q_found, phi);
    printf("  Recovered Private Key d = %lld\n\n", d);

    // Step 3: Decrypt all ciphertext blocks with recovered d
    printf("=== Decrypting ALL Intercepted Blocks ===\n");
    for (int i = 0; i < 3; i++) {
        long long decrypted = mod_exp(C[i], d, n);
        printf("  Block %d: Decrypted = %-10lld | Original = %-10lld (Match: %s)\n", 
               i + 1, decrypted, M[i], (decrypted == M[i]) ? "YES" : "NO");
    }

    printf("\n=== Answer to Question ===\n");
    printf("Question: Does knowing that one plaintext block has a common factor with n help us?\n\n");
    printf("Answer:\n");
    printf("YES, it completely breaks the cryptosystem.\n");
    printf("Since n = p * q (product of two distinct primes), any non-trivial common factor\n");
    printf("between a plaintext block M and n must be either p or q.\n");
    printf("Because C = M^e (mod n), gcd(C, n) directly yields that prime factor in O(log n) time.\n");
    printf("Once p is discovered, q = n / p, phi(n) = (p - 1)(q - 1), and the private key\n");
    printf("d = e^(-1) mod phi(n) is immediately recovered, enabling decryption of ALL blocks.\n");

    return 0;
}
