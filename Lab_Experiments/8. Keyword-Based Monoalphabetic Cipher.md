## 8. Keyword-Based Monoalphabetic Cipher

## CODE

#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main()
{

    char key[50];
    char cipher[26];
    int used[26] = {0};
    int k = 0;

    printf("Enter keyword: ");
    scanf("%s", key);

    for (int i = 0; key[i] != '\0'; i++)
    {
        char ch = toupper(key[i]);

        if (ch >= 'A' && ch <= 'Z' && !used[ch - 'A'])
        {
            cipher[k++] = ch;
            used[ch - 'A'] = 1;
        }
    }

    for (char ch = 'A'; ch <= 'Z'; ch++)
    {
        if (!used[ch - 'A'])
            cipher[k++] = ch;
    }

    printf("\nPlain : ABCDEFGHIJKLMNOPQRSTUVWXYZ\n");
    printf("Cipher: ");

    for (int i = 0; i < 26; i++)
        printf("%c", cipher[i]);

    return 0;
}
