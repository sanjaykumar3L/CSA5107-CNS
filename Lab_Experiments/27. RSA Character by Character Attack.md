# 27. RSA Character by Character Attack

## Program

#include <stdio.h>
#include <string.h>

long long mod_exp(long long base, long long exp, long long mod) {
    long long res = 1;
    base %= mod;
    while (exp > 0) {
        if (exp % 2 == 1) res = (__int128)res * base % mod;
        base = (__int128)base * base % mod;
        exp /= 2;
    }
    return res;
}

int main() {
    long long n = 99963001; // Large modulus (product of two large primes)
    long long e = 65537;

    char message[] = "ATTACK";
    int len = strlen(message);
    long long ciphertexts[len];

    printf("=== RSA Character-by-Character Attack Demonstration ===\n");
    printf("Public Key: e = %lld, n = %lld\n", e, n);
    printf("Alice encrypts message \"%s\" character by character (A=0 ... Z=25).\n\n", message);

    printf("Intercepted Ciphertexts:\n");
    for (int i = 0; i < len; i++) {
        int m = message[i] - 'A';
        ciphertexts[i] = mod_exp(m, e, n);
        printf("  '%c' (%2d) -> %lld\n", message[i], m, ciphertexts[i]);
    }

    // Attacker mounts codebook / dictionary attack:
    printf("\nStep 1: Attacker builds precomputed lookup table for all 26 possible values:\n");
    long long codebook[26];
    for (int m = 0; m < 26; m++) {
        codebook[m] = mod_exp(m, e, n);
    }
    printf("  Computing c = m^e mod n for m = 0 to 25... Done (26 entries).\n\n");

    printf("Step 2: Table lookup recovery:\n");
    char recovered[len + 1];
    for (int i = 0; i < len; i++) {
        for (int m = 0; m < 26; m++) {
            if (ciphertexts[i] == codebook[m]) {
                recovered[i] = 'A' + m;
                printf("  Ciphertext %-10lld -> Plaintext '%c' (%d)\n", ciphertexts[i], recovered[i], m);
                break;
            }
        }
    }
    recovered[len] = '\0';
    printf("\nRecovered Message: \"%s\" (Match: %s)\n", recovered, strcmp(recovered, message) == 0 ? "YES" : "NO");

    printf("\n=== Security Analysis ===\n");
    printf("Question: Is encrypting characters individually with RSA secure?\n");
    printf("Answer: NO! The small message space (26 characters) allows a trivial chosen-plaintext\n");
    printf("codebook attack. The attacker precomputes 26 ciphertexts and decrypts in O(1) time.\n");

    return 0;
}
