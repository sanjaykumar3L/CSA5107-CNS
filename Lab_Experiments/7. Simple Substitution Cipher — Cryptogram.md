## 7. Simple Substitution Cipher — Cryptogram

## CODE

#include <stdio.h>
#include <string.h>

int main()
{

    char cipher[1000];
    char map[128] = {0};

    printf("Enter ciphertext:\n");
    fgets(cipher, sizeof(cipher), stdin);

    /*
       Enter the substitutions discovered during
       frequency and word analysis.
       Example:
       map['‡'] = 'e';
       map['8'] = 't';
    */

    printf("\nDecrypted message:\n");

    for (int i = 0; cipher[i] != '\0'; i++)
    {
        if (map[(unsigned char)cipher[i]] != 0)
            printf("%c", map[(unsigned char)cipher[i]]);
        else
            printf("%c", cipher[i]);
    }

    return 0;
}
