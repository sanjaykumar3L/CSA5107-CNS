# 35. Vigenere One-Time Pad

## Program

#include <stdio.h>
#include <string.h>

int main() {
    char pt[] = "SENDMOREMONEY";
    int len = strlen(pt);
    int key[] = {9, 0, 1, 7, 23, 15, 21, 14, 11, 11, 2, 8, 9};

    char ct[len + 1];
    char dec[len + 1];

    printf("=== Vigenere One-Time Pad Demonstration ===\n");
    printf("Plaintext : \"%s\" (Length: %d)\n", pt, len);
    printf("Random Key: [");
    for (int i = 0; i < len; i++) printf("%2d%s", key[i], i < len - 1 ? " " : "");
    printf("]\n\n");

    printf("Encryption Process:\n");
    for (int i = 0; i < len; i++) {
        int p_val = pt[i] - 'A';
        int c_val = (p_val + key[i]) % 26;
        ct[i] = 'A' + c_val;
        if (i < 3) {
            printf("  P: %c(%2d) + K(%2d) mod 26 = %c(%2d)\n", pt[i], p_val, key[i], ct[i], c_val);
        }
    }
    printf("  ...\n");
    ct[len] = '\0';
    printf("Ciphertext: \"%s\"\n\n", ct);

    // Decryption
    for (int i = 0; i < len; i++) {
        int c_val = ct[i] - 'A';
        int p_val = (c_val - key[i] + 26) % 26;
        dec[i] = 'A' + p_val;
    }
    dec[len] = '\0';
    printf("Decryption with same OTP key:\n");
    printf("  Recovered Plaintext: \"%s\" (Match: %s)\n\n", dec, strcmp(pt, dec) == 0 ? "YES" : "NO");

    // Part b: Find key so that ciphertext decrypts to "CASHNOTNEEDED"
    char target[] = "CASHNOTNEEDED";
    int new_key[len];
    char dec_target[len + 1];

    for (int i = 0; i < len; i++) {
        int c_val = ct[i] - 'A';
        int t_val = target[i] - 'A';
        new_key[i] = (c_val - t_val + 26) % 26;
        dec_target[i] = 'A' + ((c_val - new_key[i] + 26) % 26);
    }
    dec_target[len] = '\0';

    printf("Part (b): Finding Key so Ciphertext decrypts to \"CASHNOTNEEDED\":\n");
    printf("  Required Key: [");
    for (int i = 0; i < len; i++) printf("%2d%s", new_key[i], i < len - 1 ? " " : "");
    printf("]\n");
    printf("  Decrypted with new key: \"%s\" (Match: %s)\n", dec_target, strcmp(target, dec_target) == 0 ? "YES" : "NO");
    printf("Property: One-Time Pad exhibits perfect secrecy!\n");

    return 0;
}
