# 17. DES Decryption Key Schedule

## Program

#include <stdio.h>
#include <stdint.h>

// Shift schedule for standard DES encryption
const int ENC_SHIFTS[16] = {1, 1, 2, 2, 2, 2, 2, 2, 1, 2, 2, 2, 2, 2, 2, 1};

// Shift schedule for direct DES decryption key generation (Right Shifts)
// Decryption uses K16, K15, ..., K1.
// After obtaining (C16, D16) by applying 28 total left shifts,
// K16 requires 0 shift. Subsequent keys are obtained by right shifting.
const int DEC_RIGHT_SHIFTS[16] = {0, 1, 2, 2, 2, 2, 2, 2, 1, 2, 2, 2, 2, 2, 2, 1};

// 28-bit circular left shift
uint32_t rotl28(uint32_t val, int shift) {
    val &= 0x0FFFFFFF;
    return ((val << shift) | (val >> (28 - shift))) & 0x0FFFFFFF;
}

// 28-bit circular right shift
uint32_t rotr28(uint32_t val, int shift) {
    val &= 0x0FFFFFFF;
    return ((val >> shift) | (val << (28 - shift))) & 0x0FFFFFFF;
}

int main() {
    // Example initial 56-bit key split into two 28-bit halves (C0, D0) after PC-1
    uint32_t C0 = 0x0F0F0F0;
    uint32_t D0 = 0x05A5A5A;

    uint32_t enc_C[16], enc_D[16];
    uint32_t dec_C[16], dec_D[16];

    printf("=== Standard DES Encryption Key Generation (Left Shifts) ===\n");
    uint32_t cur_C = C0, cur_D = D0;
    for (int i = 0; i < 16; i++) {
        cur_C = rotl28(cur_C, ENC_SHIFTS[i]);
        cur_D = rotl28(cur_D, ENC_SHIFTS[i]);
        enc_C[i] = cur_C;
        enc_D[i] = cur_D;
        printf("K%-2d | Shift: %d (Left)  | C: 0x%07X | D: 0x%07X\n", i + 1, ENC_SHIFTS[i], enc_C[i], enc_D[i]);
    }

    printf("\n=== DES Decryption Key Generation Scheme (Right Shifts) ===\n");
    // Starting with C16, D16 obtained at the end of round 16
    cur_C = enc_C[15];
    cur_D = enc_D[15];

    for (int i = 0; i < 16; i++) {
        cur_C = rotr28(cur_C, DEC_RIGHT_SHIFTS[i]);
        cur_D = rotr28(cur_D, DEC_RIGHT_SHIFTS[i]);
        dec_C[i] = cur_C;
        dec_D[i] = cur_D;
        printf("Dec Round %-2d (K%-2d) | Shift: %d (Right) | C: 0x%07X | D: 0x%07X\n", 
               i + 1, 16 - i, DEC_RIGHT_SHIFTS[i], dec_C[i], dec_D[i]);
    }

    // Verify that Decryption Keys exactly match Encryption Keys in reverse order
    int match = 1;
    for (int i = 0; i < 16; i++) {
        if (dec_C[i] != enc_C[15 - i] || dec_D[i] != enc_D[15 - i]) {
            match = 0;
            break;
        }
    }

    printf("\nVerification: %s\n", match ? "Decryption subkeys perfectly match reverse encryption subkeys!" : "Mismatch detected!");
    return 0;
}
