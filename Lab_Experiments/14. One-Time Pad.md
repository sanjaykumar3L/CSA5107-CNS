## 14. One-Time Pad — Vigenère Cipher

## CODE

#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main()
{

    char text[] = "SEND MORE MONEY";

    int key[] = {9, 0, 1, 7, 23, 15, 21, 14, 11, 11, 2, 8, 9};

    int j = 0;

    printf("Plaintext: %s\n", text);
    printf("Ciphertext: ");

    for (int i = 0; text[i] != '\0'; i++)
    {
        if (isalpha(text[i]))
        {
            char ch = toupper(text[i]);

            ch = (ch - 'A' + key[j]) % 26 + 'A';

            printf("%c", ch);
            j++;
        }
        else
        {
            printf("%c", text[i]);
        }
    }

    printf("\n");

    return 0;
}
