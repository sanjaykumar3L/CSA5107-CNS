## 11. Number of Playfair Keys

## CODE

#include <stdio.h>
#include <math.h>

int main()
{

    double keys, uniqueKeys;

    keys = tgamma(26) / tgamma(2);
    uniqueKeys = keys / (25 * 25);

    printf("Approximate number of Playfair keys:\n");
    printf("%.0e\n", keys);

    printf("\nApproximate power of 2:\n");
    printf("2^%.2f\n", log(keys) / log(2));

    printf("\nApproximate effectively unique keys:\n");
    printf("%.0e\n", uniqueKeys);

    return 0;
}
