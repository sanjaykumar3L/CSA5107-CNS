# 38. Known Plaintext Attack on Hill Cipher

## Program

#include <stdio.h>

int extended_gcd(int a, int b, int *x, int *y) {
    if (b == 0) {
        *x = 1;
        *y = 0;
        return a;
    }
    int x1, y1;
    int g = extended_gcd(b, a % b, &x1, &y1);
    *x = y1;
    *y = x1 - (a / b) * y1;
    return g;
}

int mod_inverse(int a, int m) {
    int x, y;
    int g = extended_gcd(a, m, &x, &y);
    if (g != 1) return -1;
    return (x % m + m) % m;
}

int main() {
    printf("=== Known Plaintext Attack on Hill Cipher (2x2 Matrix) ===\n");
    printf("Known Plaintext Blocks : P1 = \"FR\" (5, 17), P2 = \"ID\" (8, 3)\n");
    printf("Known Ciphertext Blocks: C1 = \"PQ\" (15, 16), C2 = \"CF\" (2, 5)\n\n");

    int P[2][2] = {{5, 8}, {17, 3}};
    int C[2][2] = {{15, 2}, {16, 5}};

    printf("Plaintext Matrix P:\n[ %d  %d ]\n[ %d %d ]\n", P[0][0], P[0][1], P[1][0], P[1][1]);
    int det = (P[0][0] * P[1][1] - P[0][1] * P[1][0]) % 26;
    if (det < 0) det += 26;
    printf("det(P) = 5*3 - 8*17 = 15 - 136 = -121 = %d mod 26\n", det);

    int det_inv = mod_inverse(det, 26);
    printf("gcd(%d, 26) = 1 (Invertible!)\n", det);

    int P_inv[2][2];
    P_inv[0][0] = (P[1][1] * det_inv) % 26;
    P_inv[0][1] = ((-P[0][1] + 26) * det_inv) % 26;
    P_inv[1][0] = ((-P[1][0] + 26) * det_inv) % 26;
    P_inv[1][1] = (P[0][0] * det_inv) % 26;

    printf("P^-1 mod 26:\n[ %d  %d ]\n[ %d  %d]\n\n", P_inv[0][0], P_inv[0][1], P_inv[1][0], P_inv[1][1]);

    // K = P_inv * C mod 26
    int K[2][2];
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            K[i][j] = (P_inv[i][0] * C[0][j] + P_inv[i][1] * C[1][j]) % 26;
        }
    }

    printf("Recovering Key Matrix K = P^-1 * C mod 26:\n[ %d   %d ]\n[ %d   %d ]\n\n", 
           K[0][0], K[0][1], K[1][0], K[1][1]);

    printf("Chosen Plaintext Attack Advantage:\n");
    printf("Choosing P = Identity Matrix [1 0; 0 1] ('BA', 'AB') outputs K directly without inversion!\n");

    return 0;
}
