## 15. Letter Frequency Attack on Additive Cipher

## CODE

#include <stdio.h>
#include <string.h>

int main()
{

    char cipher[1000];
    int freq[26] = {0};
    int total = 0;

    printf("Enter ciphertext: ");
    fgets(cipher, sizeof(cipher), stdin);

    for (int i = 0; cipher[i] != '\0'; i++)
    {
        if (cipher[i] >= 'A' && cipher[i] <= 'Z')
        {
            freq[cipher[i] - 'A']++;
            total++;
        }
        else if (cipher[i] >= 'a' && cipher[i] <= 'z')
        {
            freq[cipher[i] - 'a']++;
            total++;
        }
    }

    printf("\nFrequency:\n");

    for (int i = 0; i < 26; i++)
        printf("%c : %d\n", 'A' + i, freq[i]);

    printf("\nPossible plaintexts:\n");

    for (int key = 0; key < 26; key++)
    {
        printf("\nKey %d: ", key);

        for (int i = 0; cipher[i] != '\0'; i++)
        {
            if (cipher[i] >= 'A' && cipher[i] <= 'Z')
                printf("%c",
                (cipher[i] - 'A' - key + 26) % 26 + 'A');

            else if (cipher[i] >= 'a' && cipher[i] <= 'z')
                printf("%c",
                (cipher[i] - 'a' - key + 26) % 26 + 'a');

            else
                printf("%c", cipher[i]);
        }
    }

    return 0;
}
