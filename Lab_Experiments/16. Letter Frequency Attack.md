# 16. Letter Frequency Attack

## Program

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define ALPHABET_SIZE 26

// Standard English letter frequencies (in percentage)
const double ENGLISH_FREQ[26] = {
    8.167, 1.492, 2.782, 4.253, 12.702, 2.228, 2.015, 6.094,
    6.966, 0.153, 0.772, 4.025, 2.406, 6.749, 7.507, 1.929,
    0.095, 5.987, 6.327, 9.056, 2.758, 0.978, 2.360, 0.150,
    1.974, 0.074
};

typedef struct {
    int shift;
    double score;
    char plaintext[1024];
} Candidate;

int compareCandidates(const void *a, const void *b) {
    Candidate *c1 = (Candidate *)a;
    Candidate *c2 = (Candidate *)b;
    if (c1->score < c2->score) return -1;
    if (c1->score > c2->score) return 1;
    return 0;
}

int main() {
    char ciphertext[1024];
    int top_n;

    printf("Enter the ciphertext: ");
    if (!fgets(ciphertext, sizeof(ciphertext), stdin)) return 0;
    ciphertext[strcspn(ciphertext, "\r\n")] = '\0';

    printf("Enter number of top possible plaintexts to view (e.g., 10): ");
    if (scanf("%d", &top_n) != 1 || top_n <= 0) top_n = 10;
    if (top_n > 26) top_n = 26;

    Candidate candidates[26];

    for (int shift = 0; shift < 26; shift++) {
        candidates[shift].shift = shift;
        int counts[26] = {0};
        int total_letters = 0;

        for (int i = 0; ciphertext[i] != '\0'; i++) {
            char ch = ciphertext[i];
            if (isalpha((unsigned char)ch)) {
                char base = isupper((unsigned char)ch) ? 'A' : 'a';
                char decrypted = (ch - base - shift + 26) % 26 + base;
                candidates[shift].plaintext[i] = decrypted;
                counts[toupper((unsigned char)decrypted) - 'A']++;
                total_letters++;
            } else {
                candidates[shift].plaintext[i] = ch;
            }
        }
        candidates[shift].plaintext[strlen(ciphertext)] = '\0';

        // Calculate Chi-Square statistic
        double chi_sq = 0.0;
        if (total_letters > 0) {
            for (int i = 0; i < 26; i++) {
                double observed = (double)counts[i];
                double expected = (ENGLISH_FREQ[i] / 100.0) * total_letters;
                chi_sq += ((observed - expected) * (observed - expected)) / (expected + 1e-6);
            }
        }
        candidates[shift].score = chi_sq;
    }

    qsort(candidates, 26, sizeof(Candidate), compareCandidates);

    printf("\n=== Top %d Most Likely Plaintexts ===\n", top_n);
    for (int i = 0; i < top_n; i++) {
        printf("[%2d] (Shift: %2d, Chi-Square: %8.2f): %s\n", 
               i + 1, candidates[i].shift, candidates[i].score, candidates[i].plaintext);
    }

    return 0;
}
