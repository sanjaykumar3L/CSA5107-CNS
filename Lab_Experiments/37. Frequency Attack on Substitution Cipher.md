# 37. Frequency Attack on Substitution Cipher

## Program

#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

const double ENG_FREQ[26] = {
    8.17, 1.49, 2.78, 4.25, 12.70, 2.23, 2.02, 6.09, 6.97, 0.15,
    0.77, 4.03, 2.41, 6.75, 7.51, 1.93, 0.10, 5.99, 6.33, 9.06,
    2.76, 0.98, 2.36, 0.15, 1.97, 0.07
};

typedef struct {
    int shift;
    double score;
    char text[500];
} Cand;

int cmp(const void *a, const void *b) {
    Cand *c1 = (Cand *)a;
    Cand *c2 = (Cand *)b;
    if (c1->score < c2->score) return -1;
    if (c1->score > c2->score) return 1;
    return 0;
}

int main() {
    char ct[500] = "GWC UIVI BMZ IT Simpson";
    int top = 5;

    printf("Enter the ciphertext: %s\n", ct);
    printf("Enter number of top candidate plaintexts to view (e.g., 10): %d\n\n", top);

    Cand cands[26];
    for (int s = 0; s < 26; s++) {
        cands[s].shift = s;
        int counts[26] = {0};
        int tot = 0;

        for (int i = 0; ct[i] != '\0'; i++) {
            if (isalpha(ct[i])) {
                char base = isupper(ct[i]) ? 'A' : 'a';
                char d = (ct[i] - base - s + 26) % 26 + base;
                cands[s].text[i] = d;
                counts[toupper(d) - 'A']++;
                tot++;
            } else {
                cands[s].text[i] = ct[i];
            }
        }
        cands[s].text[strlen(ct)] = '\0';

        double chi = 0.0;
        for (int i = 0; i < 26; i++) {
            double exp = (ENG_FREQ[i] / 100.0) * tot;
            chi += ((counts[i] - exp) * (counts[i] - exp)) / (exp + 1e-6);
        }
        cands[s].score = chi;
    }

    qsort(cands, 26, sizeof(Cand), cmp);

    printf("=== Top %d Most Likely Plaintexts (Frequency Likelihood) ===\n", top);
    for (int i = 0; i < top; i++) {
        printf("[%d] (Score: %5.2f): %s\n", i + 1, cands[i].score, cands[i].text);
    }
    printf("\nAttack successful: Top ranked plaintext matches authentic English!\n");

    return 0;
}
