# 19. 3DES CBC Mode

## Program

#include <stdio.h>
#include <string.h>
#include <stdint.h>

#define BLOCK_SIZE 8

// Simplified round permutation/Feistel mock representing block cipher core
uint64_t des_primitive(uint64_t block, uint64_t key, int encrypt) {

    // A simplified 64-bit block Feistel structure modeling DES
    uint32_t L = (uint32_t)(block >> 32);
    uint32_t R = (uint32_t)(block & 0xFFFFFFFF);
    
    for (int r = 0; r < 16; r++) {
        uint32_t k = (uint32_t)((key >> (encrypt ? (r * 4) : ((15 - r) * 4))) & 0xFFFFFFFF);
        uint32_t temp = R;
        R = L ^ (R ^ k ^ 0x5A5A5A5A);
        L = temp;
    }
    return (((uint64_t)R) << 32) | L;
}

// 3DES EDE mode (Encrypt with K1, Decrypt with K2, Encrypt with K3)
uint64_t triple_des_encrypt(uint64_t block, uint64_t k1, uint64_t k2, uint64_t k3) {

    uint64_t step1 = des_primitive(block, k1, 1);
    uint64_t step2 = des_primitive(step1, k2, 0);
    uint64_t step3 = des_primitive(step2, k3, 1);
    return step3;
}

uint64_t triple_des_decrypt(uint64_t block, uint64_t k1, uint64_t k2, uint64_t k3) {

    uint64_t step1 = des_primitive(block, k3, 0);
    uint64_t step2 = des_primitive(step1, k2, 1);
    uint64_t step3 = des_primitive(step2, k1, 0);
    return step3;
}

int main() {

    uint64_t k1 = 0x133457799BBCDFF1ULL;
    uint64_t k2 = 0x1F2E3D4C5B6A7988ULL;
    uint64_t k3 = 0xA1B2C3D4E5F60718ULL;
    uint64_t iv = 0x0123456789ABCDEFULL;

    uint64_t plaintext[2] = {
        0x4E6F772069732074ULL, // "Now is t"
        0x68652074696D6520ULL  // "he time "
    };
    uint64_t ciphertext[2];
    uint64_t decrypted[2];

    printf("=== 3DES CBC Mode Demonstration ===\n");
    printf("Initialization Vector (IV): 0x%016llX\n\n", iv);

    // CBC Encryption
    uint64_t prev = iv;
    for (int i = 0; i < 2; i++) {
        uint64_t xor_block = plaintext[i] ^ prev;
        ciphertext[i] = triple_des_encrypt(xor_block, k1, k2, k3);
        prev = ciphertext[i];
        printf("Plaintext Block  [%d]: 0x%016llX\n", i + 1, plaintext[i]);
        printf("Ciphertext Block [%d]: 0x%016llX\n\n", i + 1, ciphertext[i]);
    }

    // CBC Decryption
    prev = iv;
    for (int i = 0; i < 2; i++) {
        uint64_t dec_block = triple_des_decrypt(ciphertext[i], k1, k2, k3);
        decrypted[i] = dec_block ^ prev;
        prev = ciphertext[i];
    }

    printf("=== Verification After Decryption ===\n");
    for (int i = 0; i < 2; i++) {
        printf("Decrypted Block  [%d]: 0x%016llX (Match: %s)\n", 
               i + 1, decrypted[i], (decrypted[i] == plaintext[i]) ? "YES" : "NO");
    }

    printf("\n=== Analysis and Comparison (DES vs 3DES) ===\n");
    printf("a. For Security:\n");
    printf("   Choose 3DES. Standard DES has a 56-bit key length, vulnerable to brute-force\n");
    printf("   search. 3DES provides an effective key length of 112 (2-key) or 168 (3-key) bits.\n\n");
    printf("b. For Performance:\n");
    printf("   Choose DES. DES requires only 1/3 of the computation compared to 3DES,\n");
    printf("   which performs three full cipher operations per block.\n");

    return 0;
}
